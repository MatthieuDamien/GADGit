```bash
%runscript
    cd /app
    echo "==================================="
    echo "   Application React Quark"
    echo "   Version: ${GAD_VERSION}"
    echo "==================================="
    echo ""
    echo "Modes disponibles:"
    echo "  dev   : yarn start (developpement avec hot-reload)"
    echo "  build : yarn build (reconstruction)"
    echo "  serve : serve la version build sur le port 3000"
    echo ""
    echo "Utilisation: singularity run container.sif [dev|build|serve]"
    echo "Par defaut: serve"
    echo ""

    case "$1" in
        "dev")
            echo "Demarrage en mode developpement..."
            echo "Application accessible sur http://localhost:3000"
            yarn start
            ;;
        "build") 
            echo "Construction de l'application..."
            yarn build
            echo "Build termine. Fichiers dans /app/build"
            ;;
        "serve")
            echo "Service de l'application construite..."
            echo "Application accessible sur http://localhost:3000"
            npx serve -s build -p 3000
            ;;
        *)
            echo "Demarrage par defaut: dev mode"
            echo "Application accessible sur http://localhost:3000"
            yarn start
            ;;
    esac
```
